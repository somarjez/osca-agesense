**Goal:** Add per-senior risk driver explanations (section-level + feature-level) to the senior profile and a global model insights chart to the cluster report, persisted in `ml_results.xai_data` without SHAP.

**Architecture:** Feature importance × deviation from cluster mean, computed inside `inference_service.py` after GBR predictions. Cluster feature means pre-computed once in `cluster_feature_means.json`. XAI stored as JSON in `ml_results.xai_data`. Senior profile reads from the already-loaded record — no extra HTTP calls. Cluster report fetches global importance via a new `/model_insights` endpoint proxied by `XaiController`.

**Tech Stack:** Python (numpy, scikit-learn feature_importances_), Flask, Laravel Blade, Alpine.js, Chart.js.

```python
def _compute_xai(
    feature_map: Dict[str, Any],
    named_id: int,
    ml_risk_feature_names: List[str],
) -> Dict[str, Any]:
    """
    Compute per-domain XAI explanation using feature importance x deviation.
    Returns section_drivers (top 3) and feature_drivers (top 5) per domain.
    """
    means_data = _load_cluster_feature_means()
    cluster_means: Dict[str, float] = {}
    if means_data:
        cluster_key = str(named_id)
        cluster_means = means_data.get("cluster_means", {}).get(
            cluster_key,
            means_data.get("global_mean", {})
        )
    else:
        # Fallback: 0.5 neutral midpoint
        cluster_means = {f: 0.5 for f in ml_risk_feature_names}

    def _domain_xai(model_key: str) -> Dict[str, Any]:
        gbr = _load_model(model_key)
        if gbr is None or not hasattr(gbr, "feature_importances_"):
            return {"section_drivers": [], "feature_drivers": []}

        importances = gbr.feature_importances_   # shape: (52,)
        n = len(ml_risk_feature_names)

        # Compute signed contributions
        contribs: List[Dict[str, Any]] = []
        for i, feat in enumerate(ml_risk_feature_names):
            if i >= len(importances):
                continue
            imp   = float(importances[i])
            val   = float(feature_map.get(feat, 0.0))
            mean  = float(cluster_means.get(feat, 0.5))
            contrib = imp * (val - mean)
            contribs.append({
                "feature":   feat,
                "importance": round(imp, 5),
                "value":      round(val, 4),
                "mean":       round(mean, 4),
                "raw_contrib": contrib,
            })

        # Normalise to percentage
        total_abs = sum(abs(c["raw_contrib"]) for c in contribs) or 1.0
        for c in contribs:
            c["contribution_pct"] = round(abs(c["raw_contrib"]) / total_abs * 100, 2)
            c["direction"] = "up" if c["raw_contrib"] >= 0 else "down"

        # Top-5 feature drivers
        top_feats = sorted(contribs, key=lambda x: x["contribution_pct"], reverse=True)[:5]
        feature_drivers = [
            {
                "feature":          c["feature"],
                "contribution_pct": c["contribution_pct"],
                "direction":        c["direction"],
                "value":            c["value"],
                "mean":             c["mean"],
                "importance":       c["importance"],
            }
            for c in top_feats
        ]

        # Aggregate into section drivers
        section_totals: Dict[str, float] = {}
        section_direction_map: Dict[str, float] = {}
        for c in contribs:
            sec = _FEATURE_TO_SECTION.get(c["feature"], "Other")
            section_totals[sec] = section_totals.get(sec, 0.0) + c["contribution_pct"]
            section_direction_map[sec] = section_direction_map.get(sec, 0.0) + c["raw_contrib"]

        section_total_abs = sum(section_totals.values()) or 1.0
        top_sections = sorted(section_totals.items(), key=lambda x: x[1], reverse=True)[:3]
        section_drivers = [
            {
                "section":          name,
                "contribution_pct": round(pct, 2),
                "direction":        "up" if section_direction_map.get(name, 0) >= 0 else "down",
            }
            for name, pct in top_sections
        ]

        return {"section_drivers": section_drivers, "feature_drivers": feature_drivers}

    return {
        "ic":   _domain_xai("gbr_ic_risk.pkl"),
        "env":  _domain_xai("gbr_env_risk.pkl"),
        "func": _domain_xai("gbr_func_risk.pkl"),
        "cluster_named_id": named_id,
        "computed_at": __import__("datetime").datetime.utcnow().strftime("%Y-%m-%dT%H:%M:%S"),
    }


@lru_cache(maxsize=1)
def _build_model_insights() -> Dict[str, Any]:
    """
    Global feature importance from each GBR model. Pre-built at first call, cached.
    Returns top-15 features per domain sorted by importance descending.
    """
    means_data = _load_cluster_feature_means()
    risk_features = means_data["risk_features"] if means_data else []

    def _domain_importance(model_key: str) -> List[Dict[str, Any]]:
        gbr = _load_model(model_key)
        if gbr is None or not hasattr(gbr, "feature_importances_") or not risk_features:
            return []
        pairs = [
            {"feature": feat, "importance": round(float(imp), 5)}
            for feat, imp in zip(risk_features, gbr.feature_importances_)
        ]
        pairs.sort(key=lambda x: x["importance"], reverse=True)
        return pairs[:15]

    return {
        "ic":           _domain_importance("gbr_ic_risk.pkl"),
        "env":          _domain_importance("gbr_env_risk.pkl"),
        "func":         _domain_importance("gbr_func_risk.pkl"),
        "n_seniors":    283,
        "generated_at": __import__("datetime").datetime.utcnow().strftime("%Y-%m-%dT%H:%M:%S"),
    }
```

```php
<?php

namespace App\Http\Controllers;

use Illuminate\Http\JsonResponse;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Http;
use App\Support\XaiFeatureLabels;

class XaiController extends Controller
{
    public function modelInsights(): JsonResponse
    {
        $data = Cache::remember('xai_model_insights', 60, function () {
            $resp = Http::timeout(10)->get('http://127.0.0.1:5002/model_insights');
            if ($resp->failed()) {
                return null;
            }
            $raw = $resp->json();
            // Attach human-readable labels
            foreach (['ic', 'env', 'func'] as $domain) {
                foreach ($raw[$domain] ?? [] as &$item) {
                    $item['label'] = XaiFeatureLabels::label($item['feature']);
                }
                unset($item);
            }
            return $raw;
        });

        if ($data === null) {
            return response()->json(['error' => 'Model insights unavailable'], 503);
        }

        return response()->json($data);
    }
}
```
